local espLifecycle = { stopped = false, connections = {}, cleanups = {} }
function espLifecycle:cleanup()
    if self.stopped then return end
    self.stopped = true
    for _, connection in ipairs(self.connections) do
        pcall(function() connection:Disconnect() end)
    end
    for _, cleanup in ipairs(self.cleanups) do pcall(cleanup) end
    table.clear(self.connections)
    table.clear(self.cleanups)
end


-- Canvas offsets are local GUI units, not screen pixels. Measure actual child
-- bounds so UIScale, grid wrapping, hidden entries and nested pages agree.
function espLifecycle:bindScrollCanvas(scroll, extraBottom)
    local stopped, queued = false, false
    local links, childLinks = {}, {}
    local function connect(signal, fn)
        local link = signal:Connect(fn)
        links[#links + 1] = link
        return link
    end
    local function effectiveScale()
        local factor, node = 1, scroll
        while node and not node:IsA("LayerCollector") do
            for _, object in ipairs(node:GetChildren()) do
                if object:IsA("UIScale") then factor = factor * object.Scale end
            end
            node = node.Parent
        end
        return math.max(factor, 0.001)
    end
    local function update()
        queued = false
        if stopped or espLifecycle.stopped or not scroll.Parent then return end
        local factor = effectiveScale()
        local top, bottom
        for _, child in ipairs(scroll:GetChildren()) do
            if child:IsA("GuiObject") and child.Visible then
                local y = child.AbsolutePosition.Y
                top = top and math.min(top, y) or y
                bottom = bottom and math.max(bottom, y + child.AbsoluteSize.Y)
                    or y + child.AbsoluteSize.Y
            end
        end
        local padding = scroll:FindFirstChildOfClass("UIPadding")
        local paddingHeight = 0
        if padding then
            paddingHeight = padding.PaddingTop.Offset + padding.PaddingBottom.Offset
                + (padding.PaddingTop.Scale + padding.PaddingBottom.Scale)
                    * scroll.AbsoluteSize.Y / factor
        end
        local height = math.ceil((top and (bottom - top) / factor or 0)
            + paddingHeight + (extraBottom or 8))
        -- Avoid feeding subpixel layout noise back into the canvas every frame.
        if scroll.CanvasSize.Y.Offset ~= height or scroll.CanvasSize.Y.Scale ~= 0 then
            scroll.CanvasSize = UDim2.fromOffset(0, height)
        end
    end
    local function schedule()
        if queued or stopped or espLifecycle.stopped then return end
        queued = true
        task.defer(update)
    end
    local function unwatch(child)
        local connections = childLinks[child]
        if connections then
            for _, connection in ipairs(connections) do connection:Disconnect() end
            childLinks[child] = nil
        end
    end
    local function watch(child)
        if childLinks[child] then return end
        local properties
        if child:IsA("GuiObject") then
            properties = {"AbsoluteSize", "AbsolutePosition", "Visible"}
        elseif child:IsA("UIGridStyleLayout") then
            properties = {"AbsoluteContentSize"}
        elseif child:IsA("UIPadding") then
            properties = {"PaddingTop", "PaddingBottom"}
        end
        if properties then
            local connections = {}
            childLinks[child] = connections
            for _, property in ipairs(properties) do
                connections[#connections + 1] = child:GetPropertyChangedSignal(property):Connect(schedule)
            end
        end
    end
    scroll.AutomaticCanvasSize = Enum.AutomaticSize.None
    scroll.ScrollingDirection = Enum.ScrollingDirection.Y
    scroll.ScrollingEnabled = true
    scroll.Active = true
    scroll.ClipsDescendants = true
    scroll.ElasticBehavior = Enum.ElasticBehavior.WhenScrollable
    scroll.VerticalScrollBarInset = Enum.ScrollBarInset.Always
    for _, child in ipairs(scroll:GetChildren()) do watch(child) end
    connect(scroll.ChildAdded, function(child) watch(child); schedule() end)
    connect(scroll.ChildRemoved, function(child) unwatch(child); schedule() end)
    connect(scroll:GetPropertyChangedSignal("AbsoluteSize"), schedule)
    connect(scroll:GetPropertyChangedSignal("Visible"), schedule)
    local node = scroll
    while node and not node:IsA("LayerCollector") do
        for _, object in ipairs(node:GetChildren()) do
            if object:IsA("UIScale") then
                connect(object:GetPropertyChangedSignal("Scale"), schedule)
            end
        end
        node = node.Parent
    end
    self.cleanups[#self.cleanups + 1] = function()
        stopped = true
        for _, link in ipairs(links) do link:Disconnect() end
        for child in pairs(childLinks) do unwatch(child) end
    end
    schedule()
    return schedule
end

-- ============================================================
-- HALLOWEEN 2026: AUTO BUY TIERS
-- Independent opt-in toggle in Auto Farm; one tier request per two seconds.
-- FireServer only sends the request. Currency, price and pass limits are server-owned.
-- ============================================================
do
    local elapsed, wasEnabled, lastWarning = 0, false, -math.huge
    local function resolveTierEvent()
        local node = ReplicatedStorage
        for _, name in ipairs({"Remotes", "Events", "Halloween2026Remotes", "BuyBattlePassTiers"}) do
            node = node and node:FindFirstChild(name)
            if not node then return nil end
        end
        return node:IsA("RemoteEvent") and node or nil
    end

    table.insert(espLifecycle.connections, RunService.Heartbeat:Connect(function(dt)
        if espLifecycle.stopped or not ST.auto_buy_tiers then
            elapsed, wasEnabled = 0, false
            return
        end
        if not wasEnabled then
            wasEnabled = true
            elapsed = 2 -- First request on the next heartbeat after enabling.
        end
        elapsed = elapsed + dt
        if elapsed < 2 then return end
        -- Never catch up with a burst after lag or app suspension.
        elapsed = 0
        local event = resolveTierEvent()
        if not event then
            if os.clock() - lastWarning >= 30 then
                lastWarning = os.clock()
                if notify then
                    pcall(notify, "Auto Buy Tiers", "Halloween 2026 remote not found; waiting")
                end
            end
            return
        end
        local ok, err = pcall(function() event:FireServer(1) end)
        if not ok and os.clock() - lastWarning >= 30 then
            lastWarning = os.clock()
            warn("[DaDa Auto Buy Tiers] Request failed: " .. tostring(err))
        end
    end))
    table.insert(espLifecycle.cleanups, function()
        ST.auto_buy_tiers = false
        elapsed, wasEnabled = 0, false
    end)
end

local function initRoles()
    local round_mod=nil
    get_round=function()
        if round_mod then return round_mod end
        local ok,m=pcall(function() return require(ReplicatedStorage:WaitForChild("Modules"):WaitForChild("CurrentRoundClient")) end)
        if ok and type(m)=="table" then round_mod=m end
        return round_mod
    end
    get_roles=function()
        local out={}
        local m=get_round(); local d=m and m.PlayerData
        if type(d)=="table" then for name,info in pairs(d) do if type(info)=="table" and not info.Dead then out[name]=info.Role end end end
        for _,p in ipairs(Players:GetPlayers()) do
            local c=p.Character; local bp=p:FindFirstChildOfClass("Backpack")
            local hasGun=(c and c:FindFirstChild("Gun")) or (bp and bp:FindFirstChild("Gun"))
            local hasKnife=(c and c:FindFirstChild("Knife")) or (bp and bp:FindFirstChild("Knife"))
            if hasGun and out[p.Name]~="Murderer" then out[p.Name]="Sheriff" end
            if hasKnife then out[p.Name]="Murderer" end
        end
        return out
    end
    my_role=function()
        local m=get_round(); local d=m and m.PlayerData; local me=d and d[LP.Name]
        return me and me.Role or nil
    end
    holds=function(c,n) return c~=nil and c:FindFirstChild(n)~=nil end
    lp_has_gun=function() return holds(LP.Character,"Gun") or holds(LP:FindFirstChildOfClass("Backpack"),"Gun") end
    get_hrp=function() local c=LP.Character; return c and c:FindFirstChild("HumanoidRootPart") end
    has_knife=function()
        local c=LP.Character
        if c and c:FindFirstChild("Knife") then return true end
        local bp=LP:FindFirstChildOfClass("Backpack")
        if bp and bp:FindFirstChild("Knife") then return true end
        return false
    end
    ping_sec=function()

    local function snap_push(now, pos)

    local function fit_velocity()
    	if snap_n < 3 then return nil end
    	local newest = snap_get(0)
    	local used = 0
    	local sum_d = 0
    	local win = sample_span() * 4
    	for k = 0, snap_n - 1 do
    		local t = snap_get(k)
    		if newest - t > win then break end
    		used = used + 1
    		sum_d = sum_d + t - newest
    	end
    	if used < 3 then return nil end
    	local mean_d = sum_d / used
    	local num = Vector3.zero
    	local den = 0
    	for k = 0, used - 1 do
    		local t, p = snap_get(k)
    		local d = t - newest - mean_d
    		num = num + p * d
    		den = den + d * d
    	end
    	if den < 1e-8 then return nil end
    	return num / den, -mean_d
    end

    local function recent_velocity()
    	if snap_n < 2 then return nil end
    	local newest, head = snap_get(0)
    	local fallback, fallback_age = nil, nil
    	local target_span = sample_span() * 2
    	local max_span = target_span * 2
    	for k = 1, snap_n - 1 do
    		local t, p = snap_get(k)
    		local dt = newest - t
    		if dt > max_span then break end
    		if dt > 0 then
    			fallback = (head - p) / dt
    			fallback_age = dt * 0.5
    			if dt >= target_span then
    				return fallback, fallback_age
    			end
    		end
    	end
    	return fallback, fallback_age
    end

    local KIN = {
    	ok = false,

    local function kin_clear()
    	KIN.ok = false
    	KIN.ax = 0
    	KIN.az = 0
    	KIN.smax = 0
    end

    local function fit_kin()
    	if snap_n < 5 then return nil end
    	local t0 = snap_get(0)
    	local win = math.max(sample_span() * 5, 0.12)
    	local scale = win
    	local n, s1, s2, s3, s4 = 0, 0, 0, 0, 0
    	local bx0, bx1, bx2 = 0, 0, 0
    	local bz0, bz1, bz2 = 0, 0, 0
    	for k = 0, snap_n - 1 do
    		local t, p = snap_get(k)
    		local age = t0 - t
    		if age > win then break end
    		local u = -age / scale
    		local u2 = u * u
    		n = n + 1
    		s1 = s1 + u
    		s2 = s2 + u2
    		s3 = s3 + u2 * u
    		s4 = s4 + u2 * u2
    		bx0 = bx0 + p.X
    		bx1 = bx1 + p.X * u
    		bx2 = bx2 + p.X * u2
    		bz0 = bz0 + p.Z
    		bz1 = bz1 + p.Z * u
    		bz2 = bz2 + p.Z * u2
    	end
    	if n < 5 then return nil end
    	local det = n * (s2 * s4 - s3 * s3)
    		- s1 * (s1 * s4 - s3 * s2)
    		+ s2 * (s1 * s3 - s2 * s2)
    	if math.abs(det) < 1e-9 then return nil end
    	local function solve(b0, b1, b2)
    		local d1 = n * (b1 * s4 - s3 * b2)
    			- b0 * (s1 * s4 - s3 * s2)
    			+ s2 * (s1 * b2 - b1 * s2)
    		local d2 = n * (s2 * b2 - b1 * s3)
    			- s1 * (s1 * b2 - b1 * s2)
    			+ b0 * (s1 * s3 - s2 * s2)
    		return d1 / det, d2 / det
    	end
    	local cx1, cx2 = solve(bx0, bx1, bx2)
    	local cz1, cz2 = solve(bz0, bz1, bz2)
    	local vx, vz = cx1 / scale, cz1 / scale
    	local ax, az = 2 * cx2 / (scale * scale), 2 * cz2 / (scale * scale)
    	if vx ~= vx or vz ~= vz or ax ~= ax or az ~= az then return nil end
    	return Vector3.new(vx, 0, vz), Vector3.new(ax, 0, az)
    end

    local function kin_update()
    	local kv, ka = fit_kin()
    	if not kv then
    		KIN.ok = false
    		KIN.ax = 0
    		KIN.az = 0
    		return nil
    	end
    	KIN.ok = true
    	local sp = math.sqrt(kv.X * kv.X + kv.Z * kv.Z)
    	if sp > KIN.smax then
    		KIN.smax = sp
    	else
    		KIN.smax = KIN.smax * 0.985 + sp * 0.015
    	end
    	if ka and not TR.air then
    		local am = math.sqrt(ka.X * ka.X + ka.Z * ka.Z)
    		local ax, az = ka.X, ka.Z
    		if am > P.acc_max and am > 0 then
    			ax = ax * P.acc_max / am
    			az = az * P.acc_max / am
    		end
    		KIN.ax = KIN.ax * 0.5 + ax * 0.5
    		KIN.az = KIN.az * 0.5 + az * 0.5
    	else
    		KIN.ax = KIN.ax * 0.5
    		KIN.az = KIN.az * 0.5
    	end
    	return kv
    end

    local function snap_vel(k)
    	local t0, p0 = snap_get(k)
    	local t1, p1 = snap_get(k + 1)
    	local d = t0 - t1
    	if d <= 0 then return nil end
    	return (p0 - p1) / d, d
    end

    local function vert_accel()
    	if snap_n < 3 then return nil end
    	local v0, d0 = snap_vel(0)
    	local v1, d1 = snap_vel(1)
    	if not v0 or not v1 then return nil end
    	local span = (d0 + d1) * 0.5
    	if span <= 1e-4 then return nil end
    	return (v0.Y - v1.Y) / span
    end

    local function air_vy()
    	if snap_n < 2 then return nil end
    	local edge = TR.air_edge
    	if edge <= 0 then return nil end
    	local g = grav()
    	local newest, head = snap_get(0)
    	local want = sample_span() * 2
    	local best = nil
    	for k = 1, snap_n - 1 do
    		local t, p = snap_get(k)
    		if t < edge then break end
    		local dt = newest - t
    		if dt > 1e-4 then
    			best = (head.Y - p.Y) / dt - 0.5 * g * dt
    			if dt >= want then break end
    		end
    	end
    	return best
    end

    local function engine_vel(part)
    	local ok, v = pcall(function() return part.AssemblyLinearVelocity end)
    	if not ok or typeof(v) ~= "Vector3" then
    		ok, v = pcall(function() return part.Velocity end)

    	return v

    local function vel_trust(pv, ev)
    	if not pv or not ev then return 0 end
    	local ph = Vector3.new(pv.X, 0, pv.Z)
    	local eh = Vector3.new(ev.X, 0, ev.Z)
    	local pm, em = ph.Magnitude, eh.Magnitude
    	if pm < 1 and em < 1 then return 1 end
    	if pm < 1 or em < 1 then return 0 end
    	local ratio = em / pm
    	if ratio > 1.5 or ratio < 0.6 then return 0 end
    	local align = ph.Unit:Dot(eh.Unit)
    	if align < 0.7 then return 0 end
    	local a = math.clamp((align - 0.7) / 0.25, 0, 1)
    	local r = 1 - math.clamp(math.abs(ratio - 1) / 0.4, 0, 1)
    	return a * r
    end

    local function phase_velocity(v, age, air)
    	if not v then return nil end
    	local y = 0
    	if air then
    		y = v.Y - grav() * math.clamp(age or 0, 0, sample_span() * 4)
    	end
    	return Vector3.new(v.X, y, v.Z)
    end

    local function merge_vel(fit, fit_age, fast, fast_age, engine, engine_age, air)
    	local stable = phase_velocity(fit, fit_age, air)
    	local instant = phase_velocity(fast, fast_age, air)
    	local turn = 0
    	if stable and instant then
    		local sh = Vector3.new(stable.X, 0, stable.Z)
    		local ih = Vector3.new(instant.X, 0, instant.Z)
    		if sh.Magnitude > 1 and ih.Magnitude > 1 then
    			turn = math.acos(math.clamp(sh.Unit:Dot(ih.Unit), -1, 1)) / math.pi
    		end
    	end
    	local base = instant or stable
    	if not base then return Vector3.zero, 0, nil, 0 end
    	if stable and instant then
    		local agility = math.clamp(turn * 2.2, 0, 1)
    		base = stable:Lerp(instant, 0.4 + 0.6 * agility)
    	end
    	local trust = 0
    	if engine then
    		local live = phase_velocity(engine, engine_age, air)
    		trust = vel_trust(base, live)
    		if trust > 0 and air then
    			base = Vector3.new(base.X, base.Y, base.Z):Lerp(Vector3.new(base.X, live.Y, base.Z), trust * 0.35)
    		end
    	end
    	return base, turn, instant or stable, trust
    end

    	TR.spoof = 0

    local function track_fresh(now)
    	local part = target_part
    	if not part or not part.Parent then
    		TR.fresh_ok = false
    		return
    	end
    	local pos = part.Position
    	local g = grav()
    	local sv = snap_vel(0)
    	local vy = sv and sv.Y or 0
    	local accel = vert_accel()
    	local falling = accel ~= nil and accel < -g * 0.5
    	local guess = stand_clearance()
    	local reach = guess + 6 + math.abs(vy) * sample_span() * 4
    	local air
    	local gy = ground_below(pos, reach)
    	if gy then
    		local clr = pos.Y - gy
    		TR.clr = clr
    		if math.abs(vy) < 1 and not falling then
    			if GC.seen then
    				if clr < GC.base then
    					GC.base = GC.base * 0.7 + clr * 0.3
    				else
    					GC.base = GC.base * 0.98 + clr * 0.02
    				end
    			else
    				GC.base = clr
    				GC.seen = true
    			end
    		end
    		local floor = GC.seen and GC.base or guess
    		local tol = math.max(floor * 0.35, 1)
    		air = clr > floor + tol
    		if not air and falling and math.abs(vy) > 4 and clr > floor + 0.35 then
    			air = true
    		end
    	else
    		air = true
    	end
    	if air ~= TR.air then
    		TR.air_edge = now
    		if air then
    			TR.air_since = now
    			TR.jump_fresh = true
    			TR.jump_v = JL.seen and JL.v or math.max(vy, 0)
    		else
    			TR.jump_fresh = false
    			TR.jump_v = 0
    		end
    	end
    	local model_vy = TR.jump_v - g * math.max(0, now - TR.air_since)
    	TR.air = air
    	TR.jumping = air and (vy > 1 or model_vy > 1)
    end

    local function track(now)
    	local part = target_part
    	if not part or not part.Parent then
    		if TR.part then track_clear() end
    		return
    	end
    	track_fresh(now)
    	local pos = part.Position
    	if part ~= TR.part or not TR.pos then
    		track_seed(part, pos, now)
    		return
    	end
    	local dt = now - TR.time
    	if dt > 0.75 or (pos - TR.pos).Magnitude > 140 then
    		track_seed(part, pos, now)
    		return
    	end
    	if dt <= 0 then return end
    	if (pos - TR.pos).Magnitude == 0 then
    		if TR.gap > 0 and dt >= TR.gap then
    			TR.vel = Vector3.zero
    			TR.fresh = Vector3.zero
    		end
    		return
    	end
    	step_push(dt)
    	TR.gap = dt
    	snap_push(now, pos)
    	TR.pos = pos
    	TR.time = now
    	local fit, fit_age = fit_velocity()
    	local fast, fast_age = recent_velocity()
    	local engine = engine_vel(part)
    	local fresh, turn, instant, trust = merge_vel(fit, fit_age, fast, fast_age, engine, sample_span() * 0.5, TR.air)
    	local kv = kin_update()
    	if kv then
    		fresh = Vector3.new(kv.X, fresh.Y, kv.Z)
    	end
    	if engine and trust <= 0 then
    		if TR.spoof < 20 then TR.spoof = TR.spoof + 1 end
    	elseif TR.spoof > 0 then
    		TR.spoof = TR.spoof - 1
    	end
    	if TR.air then
    		local vy = air_vy()
    		if vy then
    			fresh = Vector3.new(fresh.X, vy, fresh.Z)
    			local since = math.max(0, now - TR.air_edge)
    			if TR.jump_fresh and since <= 0.2 then
    				local impulse = vy + grav() * since
    				if impulse > 1 then
    					if JL.seen then
    						JL.v = JL.v * 0.7 + impulse * 0.3
    					else
    						JL.v = impulse
    						JL.seen = true
    					end
    					if impulse > TR.jump_v then TR.jump_v = impulse end
    				end
    			else
    				TR.jump_fresh = false
    			end
    		end
    	end
    	TR.vel = fresh
    	TR.ready = fit ~= nil or fast ~= nil
    	TR.fresh = TR.vel
    	TR.fresh_ok = TR.ready
    	TR.turn = turn
    	local raw = instant or fresh
    	vel_push(now, raw.X, raw.Z)
    end

    local function raw_rtt()
    	local a, b
    	local ok, ms = pcall(function()
    		return stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    	end)
    	if ok and type(ms) == "number" and ms == ms and ms > 4 and ms < 800 then
    		a = ms / 1000
    	end
    	local fine, value = pcall(function() return lp:GetNetworkPing() end)
    	if fine and type(value) == "number" and value == value and value > 0 then
    		local rtt = value * 2
    		if rtt > 0.004 and rtt < 0.8 then b = rtt end
    	end
    	if a and b then return (a + b) * 0.5 end
    	return a or b
    end

    local function sample_ping()
    	local rtt = raw_rtt()
    	if not rtt or rtt ~= rtt then return end
    	rtt = math.clamp(rtt, 0, 1)
    	if EC.seen then
    		EC.jitter = EC.jitter * 0.9 + math.abs(rtt - EC.rtt) * 0.1
    		EC.rtt = EC.rtt * 0.82 + rtt * 0.18
    	else
    		EC.rtt = rtt
    		EC.jitter = 0
    		EC.seen = true
    	end
    	EC.ping = EC.rtt
    end

    local function lead_time()
    	if not EC.seen then return 0 end
    	local stale = 0
    	if TR.time > 0 and EC.step_seen then
    		stale = math.clamp(os.clock() - TR.time, 0, EC.step)
    	end
    	return math.clamp(EC.rtt + EC.jitter * 0.5 + stale, 0, 1)
    end

    local function rotate_y(v, ang)
    	local c, s = math.cos(ang), math.sin(ang)
    	return Vector3.new(v.X * c - v.Z * s, v.Y, v.X * s + v.Z * c)
    end

    local function dir_stats(win)
    	if SK.vn < 4 then return 1, 0 end
    	win = math.max(win, sample_span() * 3)
    	local newest = SK.vt[SK.vi]
    	local sx, sz, n = 0, 0, 0
    	local prev = nil
    	local turn, turn_n = 0, 0
    	local oldest = newest
    	for k = 0, SK.vn - 1 do
    		local idx = (SK.vi - k - 1) % P.ring + 1
    		local t = SK.vt[idx]
    		if newest - t > win then break end
    		local hx, hz = SK.dx[idx], SK.dz[idx]
    		local m = math.sqrt(hx * hx + hz * hz)
    		if m > 0 then
    			sx = sx + hx / m
    			sz = sz + hz / m
    			n = n + 1
    			local ang = math.atan2(hz, hx)
    			if prev then
    				local d = ang - prev
    				while d > math.pi do d = d - 6.2831853 end
    				while d < -math.pi do d = d + 6.2831853 end
    				turn = turn + d
    				turn_n = turn_n + 1
    			end
    			prev = ang
    			oldest = t
    		end
    	end
    	if n < 2 then return 1, 0 end
    	local coh = math.clamp(math.sqrt(sx * sx + sz * sz) / n, 0, 1)
    	local omega = 0
    	local elapsed = newest - oldest
    	if turn_n >= 1 and elapsed > 1e-3 then
    		omega = -turn / elapsed
    	end
    	return coh, omega
    end

    local function predict_from(base, sa, sb, fh, now)
    	local span = math.max(0, sa + sb)
    	local g = grav()
    	local dir = fh
    	if dir.Magnitude == 0 then
    		dir = Vector3.new(TR.vel.X, 0, TR.vel.Z)
    	end
    	local x, z
    	if span > 0 and KIN.ok then
    		local age = math.clamp(now - TR.time, 0, sample_span() * 2)
    		local ax, az = KIN.ax, KIN.az
    		if TR.air or math.sqrt(ax * ax + az * az) < P.acc_min then ax, az = 0, 0 end
    		local vx = dir.X + ax * age
    		local vz = dir.Z + az * age
    		local ta = math.min(span, P.acc_t)
    		local dx = vx * span + 0.5 * ax * ta * ta
    		local dz = vz * span + 0.5 * az * ta * ta
    		local reach = math.sqrt(dx * dx + dz * dz)
    		local cap = math.max(KIN.smax * P.speed_head, P.speed_floor) * span
    		if reach > cap and reach > 1e-6 then
    			dx = dx * cap / reach
    			dz = dz * cap / reach
    		end
    		x = base.X + dx
    		z = base.Z + dz
    	else
    		local hspan = span
    		if span > 0 and dir.Magnitude > 0 and not TR.air then
    			local coh, omega = dir_stats(span)
    			local conf = math.clamp(coh, 0, 1) * (1 - math.clamp(TR.turn, 0, 1) * 0.5)
    			if omega ~= 0 then
    				dir = rotate_y(dir, math.clamp(omega * span * 0.5 * conf, -0.6, 0.6))
    			end
    			hspan = span * (0.85 + 0.15 * conf)
    		end
    		x = base.X + dir.X * hspan
    		z = base.Z + dir.Z * hspan
    	end
    	local y = base.Y
    	if TR.air and span > 0 then
    		local vy = TR.vel.Y
    		local phase = math.max(0, now - TR.air_since)
    		local modeled = TR.jump_v - g * phase
    		if TR.jumping and TR.jump_v > 0 and g > 0 and phase <= TR.jump_v / g and modeled > vy then
    			vy = modeled
    		end
    		y = base.Y + vy * span - 0.5 * g * span * span
    		if y < base.Y then
    			local clearance = stand_clearance()
    			local reach = base.Y - y + clearance
    			local gy = ground_below(Vector3.new(x, base.Y, z), reach)
    			if gy then
    				local floor = gy + clearance
    				if y < floor then y = floor end
    			end
    		end
    	end
    	return Vector3.new(x, y, z)
    end

    local function build_hyps(base, now)
    	table.clear(HY.pos)
    	table.clear(HY.w)
    	local horizon = SL.predict and TR.ready and lead_time() or 0
    	local fh = Vector3.new(TR.fresh.X, 0, TR.fresh.Z)
    	HY.primary = predict_from(base, 0, horizon, fh, now)
    	HY.n = 1
    	HY.pos[1] = HY.primary
    	HY.w[1] = 1
    	HY.weight = 1
    	HY.stamp = now
    end

    local function score_axis(anchor, axis)
    	local covered = 0
    	local lo, hi = 0, 0
    	for k = 1, HY.n do
    		local d = HY.pos[k] - anchor
    		local a = d:Dot(axis)
    		local perp = (d - axis * a).Magnitude
    		if perp <= P.hit_r then
    			covered = covered + HY.w[k]
    			if a < lo then lo = a end
    			if a > hi then hi = a end
    		end
    	end
    	return covered, lo, hi
    end

    local function corridor_axes(anchor)
    	table.clear(axis_pool)
    	local n = 0
    	local function add(v)
    		if typeof(v) ~= "Vector3" or v.Magnitude < 1e-4 then return end
    		local u = v.Unit
    		for k = 1, n do
    			if axis_pool[k]:Dot(u) > 0.985 then return end
    		end
    		n = n + 1
    		axis_pool[n] = u
    	end
    	local fh = Vector3.new(TR.fresh.X, 0, TR.fresh.Z)
    	if TR.air then add(TR.fresh) end
    	add(fh)
    	for k = 1, HY.n do
    		add(HY.pos[k] - anchor)
    	end
    	add(TR.fresh)
    	add(Vector3.new(0, 1, 0))
    	return n
    end

    local function build_corridor(now)
    	local part = target_part
    	if not part or not part.Parent then return nil end
    	local base = part.Position
    	build_hyps(base, now)
    	local anchor = HY.primary or base
    	local count = corridor_axes(anchor)
    	local best_axis, best_cov, best_lo, best_hi = nil, -1, 0, 0
    	for k = 1, count do
    		local axis = axis_pool[k]
    		local cov, lo, hi = score_axis(anchor, axis)
    		if cov > best_cov then
    			best_axis, best_cov, best_lo, best_hi = axis, cov, lo, hi
    		end
    	end
    	if not best_axis then return nil end
    	HY.conf = HY.weight > 0 and best_cov / HY.weight or 0

    	local pad = P.pad
    	local origin = anchor + best_axis * (best_lo - pad)
    	local aim = anchor + best_axis * (best_hi + pad)
    	if (aim - origin).Magnitude < 4 then
    		origin = anchor - best_axis * 4
    		aim = anchor + best_axis * 4
    	end
    	return origin, aim, HY.conf, anchor
    end

    local pred_off = Vector3.zero
    local pred_stamp = 0

    local function lead_offset()
    	local part = target_part
    	if not part or not part.Parent then return Vector3.zero end
    	if not SL.predict or not TR.ready then return Vector3.zero end
    	local base = part.Position
    	local now = os.clock()
    	local fh = Vector3.new(TR.fresh.X, 0, TR.fresh.Z)
    	local point = predict_from(base, 0, lead_time(), fh, now)
    	local off = point - base
    	pred_stamp = now
    	pred_off = off
    	return pred_off
    end

    local hit_names={"HumanoidRootPart","UpperTorso","Torso","LowerTorso","Head","RightUpperArm","LeftUpperArm","Right Arm","Left Arm","RightUpperLeg","LeftUpperLeg","Right Leg","Left Leg","RightLowerLeg","LeftLowerLeg"}
    local hit_parts={}; local hit_count=0; local hit_char=nil
    local function refresh_parts()
        local char=target_char
        if char==hit_char then return end
        table.clear(hit_parts); hit_count=0; hit_char=char
        if not char then return end
        for k=1,#hit_names do
            local part=char:FindFirstChild(hit_names[k])
            if part and part:IsA("BasePart") then hit_count=hit_count+1; hit_parts[hit_count]=part end
        end
    end
    local function los_clear(origin,point)
        if not origin or not point then return false end
        local delta=point-origin; local dist=delta.Magnitude
        if dist<0.5 then return true end
        if dist>MAX_RANGE then return false end
        local hit=trace(origin,delta)
        if not hit then return true end
        local inst=hit.Instance
        local char=target_char
        if inst and char and (inst==char or inst:IsDescendantOf(char)) then return true end
        return (hit.Position-origin).Magnitude>=dist-0.75
    end
    local function pick_point(origin,strict)
        refresh_parts()
        if hit_count==0 then return nil end
        local off=lead_offset()
        local first=nil
        for k=1,hit_count do
            local part=hit_parts[k]
            if not part.Parent then hit_char=nil
            else
                local point=part.Position+off
                if not origin then return point end
                if not first then first=point end
                if los_clear(origin,point) then return point end
            end
        end
        if strict then return nil end
        return first
    end
    local force_att=nil; local force_saved=nil; local force_stamp=0
    local function restore_origin()
        local att=force_att
        if not att then return end
        local saved=force_saved
        force_att=nil; force_saved=nil
        if saved then pcall(function() if att.Parent then att.CFrame=saved end end) end
    end
    local function push_origin(cf)
        local att=gun_attachment()
        if not att then return false end
        if force_att and force_att~=att then restore_origin() end
        if not force_att then
            local ok,saved=pcall(function() return att.CFrame end)
            if not ok or typeof(saved)~="CFrame" then return false end
            force_att=att; force_saved=saved
        end
        force_stamp=os.clock()
        local ok=pcall(function() att.WorldCFrame=cf end)
        if not ok then restore_origin() return false end
        task.defer(restore_origin); return true
    end
    local function is_target_hit(inst)
        local char=target_char
        if not inst or not char then return false end
        return inst==char or inst:IsDescendantOf(char)
    end
    local function force_clear(origin,aim)
        local hit=trace(origin,aim-origin)
        if not hit then return false end
        return is_target_hit(hit.Instance)
    end
    local function force_velocity()
        if TR.fresh_ok and TR.fresh.Magnitude>0.5 then return TR.fresh end
        if TR.ready and TR.vel.Magnitude>0.5 then return TR.vel end
        return Vector3.zero
    end
    local function resolve_force()
        local part=target_part
        if not part or not part.Parent then return nil end
        local live=part.Position; local now=os.clock()
        local origin,aim,conf,anchor=build_corridor(now)
        if origin and aim then
            local axis=aim-origin; local span=axis.Magnitude
            if span>1e-3 then
                local u=axis/span; local mark=anchor or live
                local behind=(mark-origin):Dot(u)
                if behind<P.pad then origin=origin-u*(P.pad-behind) end
                local ahead=(aim-mark):Dot(u)
                if ahead<P.min_span then aim=mark+u*P.min_span end
                local want=SL.stand_off
                while want>0 do
                    local probe=origin-u*want
                    if (aim-probe).Magnitude<=P.max_span and los_clear(probe,mark) and los_clear(probe,live) then origin=probe; break end
                    want=want-3
                end
                if (aim-origin).Magnitude>P.max_span then origin=aim-u*P.max_span end
                return CFrame.new(origin,aim),CFrame.new(aim),conf or 0,mark
            end
        end
        local vel=force_velocity(); local dir=Vector3.new(0,-1,0)
        if vel.Magnitude>3 then dir=vel.Unit
        else
            local mine=SL_origin()
            if mine then local delta=live-mine.Position if delta.Magnitude>2 then dir=delta.Unit end end
        end
        local back=live-dir*6
        local front=live+dir*math.max(P.min_span,vel.Magnitude*lead_time()+8)
        if not force_clear(back,front) then back=live-dir*2.5 end
        return CFrame.new(back,front),CFrame.new(front),0,live
    end
    local function shot_shift(dt)
        if not SL.predict or not TR.ready or dt<=0 then return Vector3.zero end
        local shift=Vector3.new(TR.vel.X*dt,0,TR.vel.Z*dt)
        if TR.air then
            local g=grav(); local horizon=lead_time(); local vy=TR.vel.Y
            local phase=math.max(0,os.clock()-TR.air_since)
            local modeled=TR.jump_v-g*phase
            if TR.jumping and TR.jump_v>0 and g>0 and phase<=TR.jump_v/g and modeled>vy then vy=modeled end
            shift=Vector3.new(shift.X,vy*dt-g*horizon*dt-0.5*g*dt*dt,shift.Z)
        end
        return shift
    end
    local function compensate_force(origin_cf,aim_cf,started)
        local shift=shot_shift(math.max(0,os.clock()-started))
        if shift==Vector3.zero then return origin_cf,aim_cf end
        local origin=origin_cf.Position+shift; local aim=aim_cf.Position+shift
        return CFrame.new(origin,aim),CFrame.new(aim)
    end
    resolve_shot=function()
        if not SL.enabled or not SL.am_sheriff or not SL_alive() then return nil end
        if SL.force then
            local started=os.clock()
            local origin_cf,aim_cf=resolve_force()
            if origin_cf and aim_cf then
                origin_cf,aim_cf=compensate_force(origin_cf,aim_cf,started)
                if push_origin(origin_cf) then return aim_cf end
            end
        end
        local cf=SL_origin()
        local aim=pick_point(cf and cf.Position or nil,false)
        if not aim then return nil end
        return CFrame.new(aim)
    end
    local function compensate_resolve(cf)
        if SL.force or typeof(cf)~="CFrame" then return cf end
        return CFrame.new(cf.Position+shot_shift(math.max(0,os.clock()-pred_stamp)))
    end
    -- WeaponService targeting only: never intercept or suppress FireServer.
    local weapon_service=nil
    local method_hooks={}
    local function get_weapon_service()
        if weapon_service then return weapon_service end
        local services=ReplicatedStorage:FindFirstChild("ClientServices")
        local module=services and services:FindFirstChild("WeaponService")
        if not module or not module:IsA("ModuleScript") then return nil end
        local ok,m=pcall(require,module)
        if ok and type(m)=="table" then weapon_service=m end
        return weapon_service
    end
    local function try_target()
        if espLifecycle.stopped then return nil end
        -- Gun targeting stays independent of KnifeThrown.
        if not SL.enabled then return nil end
        -- Protect the entire targeting path, not only resolve_shot. Any failure
        -- must fall through to the game's original targeting function.
        local ok,cf=pcall(function()
            sample_ping()
            local aim=resolve_shot()
            if typeof(aim)~="CFrame" then return nil end
            return compensate_resolve(aim)
        end)
        if ok and typeof(cf)=="CFrame" then return cf end
        if not ok then
            SL.hook_error=tostring(cf)
            restore_origin()
        end
        return nil
    end
    local function install_method(m,key)
        local previous=m[key]
        local state=method_hooks[key]
        if state and previous==state.wrapper then return true end
        if type(previous)~="function" then return false end
        -- Capture the fallback per wrapper. Reinstallation must not overwrite a
        -- shared original-function variable and create a recursive hook chain.
        local function wrapper(...)
            local cf=try_target()
            if cf then return cf end
            return previous(...)
        end
        local ok,err=pcall(function() m[key]=wrapper end)
        if not ok then return false,tostring(err) end
        method_hooks[key]={wrapper=wrapper,original=previous}
        return m[key]==wrapper
    end
    install_hooks=function()
        if espLifecycle.stopped then return false end
        local m=get_weapon_service()
        if not m then return false end
        local was_readonly=nil
        if type(isreadonly)=="function" then
            local ok,value=pcall(isreadonly,m)
            if ok then was_readonly=value end
        end
        if type(setreadonly)=="function" then pcall(setreadonly,m,false) end
        local mouse,mouse_error=install_method(m,"GetMouseTargetCFrame")
        local screen,screen_error=install_method(m,"GetTargetPosition")
        if was_readonly==true and type(setreadonly)=="function" then pcall(setreadonly,m,true) end
        SL.hook_error=mouse_error or screen_error
        SL.hook_stats={weapon_service=true,mouse=mouse,screen=screen}
        return mouse or screen
    end
    SL_installHooks=install_hooks
    table.insert(espLifecycle.cleanups,function()
        SL.enabled=false
        restore_origin()
        track_clear()
        pred_off=Vector3.zero
        pred_stamp=0
        local m=weapon_service
        if not m then return end
        local was_readonly=nil
        if type(isreadonly)=="function" then
            local ok,value=pcall(isreadonly,m)
            if ok then was_readonly=value end
        end
        if type(setreadonly)=="function" then pcall(setreadonly,m,false) end
        for key,state in pairs(method_hooks) do
            if m[key]==state.wrapper then
                pcall(function() m[key]=state.original end)
            end
        end
        if was_readonly==true and type(setreadonly)=="function" then pcall(setreadonly,m,true) end
    end)
    local function get_gun_sh()
        local char=LP.Character
        if char then local g=char:FindFirstChild("Gun") if g then return g,true end end
        local bp=LP:FindFirstChildOfClass("Backpack")
        if bp then local g=bp:FindFirstChild("Gun") if g then return g,false end end
        return nil,false
    end
    local function fire_gun_sh(gun,start_cf,aim_cf)
        if not gun or not start_cf or not aim_cf then return false end
        local remote=gun:FindFirstChild("Shoot")
        if not remote or not remote:IsA("RemoteEvent") then return false end
        return (pcall(function() remote:FireServer(start_cf,aim_cf) end))
    end
    local function auto_step(now)
        if not SL.auto_on or not SL.enabled or not SL.am_sheriff or getgenv().AUTOFARM_HOLD or not SL_alive() then want_since=0; return end
        local gun,equipped=get_gun_sh()
        if not gun then want_since=0; return end
        if gun~=gap_gun then gap_gun=gun; gap_reset() end
        if not equipped then
            want_since=0
            local hum=LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
            if hum then pcall(function() hum:EquipTool(gun) end) end
            return
        end
        if want_since==0 then want_since=now end
        local hold=SL.auto_delay
        if hold<SL.fire_gap then hold=SL.fire_gap end
        local since=SL_lastShot>0 and SL_lastShot or SL.last_shot
        if now-since<hold then return end
        if SL.force then
            local started=os.clock()
            local origin_cf,aim_cf=resolve_force()
            if not origin_cf or not aim_cf then return end
            origin_cf,aim_cf=compensate_force(origin_cf,aim_cf,started)
            if fire_gun_sh(gun,origin_cf,aim_cf) then SL.last_shot=now; SL_lastShot=now end
            return
        end
        local cf=SL_origin()
        if not cf then return end
        local aim=pick_point(cf.Position,true)
        if not aim then return end
        local aim_cf=compensate_resolve(CFrame.new(aim))
        if fire_gun_sh(gun,cf,aim_cf) then SL.last_shot=now; SL_lastShot=now end
    end
    -- Match the source predictor's Heartbeat cadence: ping must be sampled
    -- before tracking/auto-fire, even when no mouse shot has been requested.
    local prediction_running=false
    local function prediction_step(now)
        if espLifecycle.stopped then return end
        if not SL.enabled then
            if prediction_running then
                track_clear()
                pred_off=Vector3.zero
                pred_stamp=0
                prediction_running=false
            end
            return
        end
        prediction_running=true
        sample_ping()
        track(now)
    end
    _G.DaDa_SL_prediction_step=prediction_step
    _G.DaDa_SL_auto_step=auto_step
    _G.DaDa_SL_track=track
    _G.DaDa_SL_restore_origin=restore_origin
    _G.DaDa_SL_target_alive=SL_alive
end

local function initGetGun()
    local function find_gun_drop()
        for _,o in ipairs(workspace:GetDescendants()) do if o.Name=="GunDrop" and o:IsA("BasePart") then return o end end
        return nil
    end
    local function grab_touch(hrp,gun) pcall(function() if firetouchinterest then firetouchinterest(hrp,gun,0); firetouchinterest(hrp,gun,1) else gun.CFrame=hrp.CFrame end end) end
    local function grab_teleport(hrp,gun) pcall(function() gun.CFrame=hrp.CFrame end) end
    local function grab_player_tp(hrp,gun)
        if my_role()=="Murderer" then return end
        if ST.teleporting then return end
        ST.teleporting=true
        task.spawn(function()
            local old=hrp.CFrame
            pcall(function() hrp.CFrame=gun.CFrame; task.wait(0.08); local h=get_hrp(); if h and ST.getgun_tp_back then h.CFrame=old end end)
            task.wait(0.2); ST.teleporting=false
        end)
    end
    local function grab_prompt(gun)
        if has_knife() then return end
        if not my_role() then return end
        local hrp=get_hrp(); if not hrp then return end
        pcall(function() gun.CFrame=hrp.CFrame end)
        local p=gun:FindFirstChildOfClass("ProximityPrompt")
        if p and fireproximityprompt then pcall(function() fireproximityprompt(p) end) end
    end
    getgun_step=function(now)
        if not ST.getgun then return end
        if now-ST.getgun_last<ST.getgun_interval then return end
        ST.getgun_last=now
        local hrp=get_hrp(); if not hrp then return end
        local gun=find_gun_drop(); if not gun then return end
        if ST.getgun_mode=="Touch" then grab_touch(hrp,gun)
        elseif ST.getgun_mode=="Teleport" then grab_teleport(hrp,gun)
        elseif ST.getgun_mode=="PlayerTP" then grab_player_tp(hrp,gun)
        elseif ST.getgun_mode=="AutoGrab" then grab_prompt(gun) end
    end
end

-- Independent runtime profiles. The role checkboxes select EDIT targets, not render targets.
do
    local api={order={"Urself","Murder","Sheriff","Innocent"},profiles={},selected={},preview="Murder"}
    api.fields={"enabled","esp_box","esp_name","esp_dist","esp_health","esp_skel","esp_line",
        "esp_chams","esp_outline","esp_cube_v1","esp_cube_v2","esp_solid_body","esp_fill_chams",
        "esp_flags","esp_max_dist","esp_color"}
    api.colors={Red=Color3.fromRGB(255,90,90),Blue=Color3.fromRGB(90,160,255),
        Green=Color3.fromRGB(80,255,140),White=Color3.fromRGB(230,230,240),
        Purple=Color3.fromRGB(190,90,255),Gold=Color3.fromRGB(255,190,65),Cyan=Color3.fromRGB(0,220,255)}
    for _,role in ipairs(api.order) do
        local profile={enabled=true,esp_color="Role"}
        for _,key in ipairs(api.fields) do
            if key~="enabled" and key~="esp_color" then profile[key]=ST[key] end
        end
        profile.esp_flags=profile.esp_flags==true
        api.profiles[role]=profile; api.selected[role]=true
    end
    function api.role(player,roles)
        if player==LP then return "Urself" end
        local role=roles[player.Name]
        if role=="Murderer" or role=="Murder" then return "Murder" end
        if role=="Sheriff" or role=="Hero" then return "Sheriff" end
        return "Innocent"
    end
    function api.profile(player,roles) return api.profiles[api.role(player,roles)] end
    function api.color(role,profile)
        local value=profile.esp_color
        if type(value)=="string" and value:match("^#%x%x%x%x%x%x$") then return Color3.fromHex(value) end
        if api.colors[value] then return api.colors[value] end
        if role=="Murder" then return api.colors.Red end
        if role=="Sheriff" then return api.colors.Blue end
        if role=="Urself" then return api.colors.Cyan end
        return api.colors.White
    end
    function api.editRole()
        if api.selected[api.preview] then return api.preview end
        for _,role in ipairs(api.order) do if api.selected[role] then return role end end
        return api.preview
    end
    function api.read(key) return api.profiles[api.editRole()][key] end
    function api.write(key,value)
        for _,role in ipairs(api.order) do
            if api.selected[role] then api.profiles[role][key]=value end
        end
    end
    ST._espRoles=api
end

local function initESP()
    local SKEL_R6={{"Head","Torso"},{"Torso","Left Arm"},{"Torso","Right Arm"},{"Torso","Left Leg"},{"Torso","Right Leg"}}
    local SKEL_R15={{"Head","UpperTorso"},{"UpperTorso","LowerTorso"},{"UpperTorso","LeftUpperArm"},{"LeftUpperArm","LeftLowerArm"},{"LeftLowerArm","LeftHand"},{"UpperTorso","RightUpperArm"},{"RightUpperArm","RightLowerArm"},{"RightLowerArm","RightHand"},{"LowerTorso","LeftUpperLeg"},{"LeftUpperLeg","LeftLowerLeg"},{"LeftLowerLeg","LeftFoot"},{"LowerTorso","RightUpperLeg"},{"RightUpperLeg","RightLowerLeg"},{"RightLowerLeg","RightFoot"}}
    local highlights = {}
    local api=ST._espRoles
    local volumes={}
    local solidGui, solidViewport, solidWorld, solidCamera
    local function ensureSolidViewport()
        if solidGui and solidGui.Parent then return end
        solidGui=Instance.new("ScreenGui")
        solidGui.Name="DaDaSolidBodyESP"; solidGui.IgnoreGuiInset=true
        solidGui.ResetOnSpawn=false; solidGui.DisplayOrder=0
        pcall(function() solidGui.Parent=CoreGui end)
        if not solidGui.Parent then solidGui.Parent=LP:WaitForChild("PlayerGui") end
        solidViewport=Instance.new("ViewportFrame")
        solidViewport.Name="RoleBody"; solidViewport.Size=UDim2.fromScale(1,1)
        solidViewport.BackgroundTransparency=1; solidViewport.BorderSizePixel=0
        solidViewport.Ambient=Color3.new(1,1,1); solidViewport.LightColor=Color3.new(0,0,0)
        solidViewport.Active=false; solidViewport.Parent=solidGui
        solidWorld=Instance.new("WorldModel"); solidWorld.Parent=solidViewport
        solidCamera=Instance.new("Camera"); solidCamera.Parent=solidViewport
        solidViewport.CurrentCamera=solidCamera
    end
    local function clearSolidViewport()
        if solidGui then solidGui:Destroy() end
        solidGui=nil; solidViewport=nil; solidWorld=nil; solidCamera=nil
    end
    local function dropVolumes(player)
        local entry=volumes[player]; volumes[player]=nil
        if not entry then return end
        for _,object in ipairs(entry.objects) do pcall(function() object:Destroy() end) end
    end
    local function clearVolumes()
        for player in pairs(volumes) do dropVolumes(player) end
        clearSolidViewport()
    end
    local function bodyParts(char)
        local parts={}
        -- Rig parts only: no root, tools, accessory handles or weapon geometry.
        for _,part in ipairs(char:GetChildren()) do
            if part:IsA("BasePart") and part.Name~="HumanoidRootPart" then
                table.insert(parts,part)
            end
        end
        return parts
    end
    local function newBox(entry,part,size,offset,transparency)
        local box=Instance.new("BoxHandleAdornment")
        table.insert(entry.objects,box)
        box.Name="DaDaESPCube"; box.Adornee=part; box.Size=size
        box.CFrame=offset or CFrame.new(); box.AlwaysOnTop=true; box.ZIndex=5
        box.Transparency=transparency; box.Parent=part
        table.insert(entry.boxes,box)
        return box
    end
    local function newSolidPart(entry,part)
        local archivable=part.Archivable
        local ok,copy=pcall(function()
            part.Archivable=true
            return part:Clone()
        end)
        pcall(function() part.Archivable=archivable end)
        if not ok or not copy then
            warn("[DaDa Solid Body] Cannot clone "..part.Name..": "..tostring(copy))
            return
        end
        table.insert(entry.objects,copy)
        -- Remove scripts, joints, decals, clothing and PBR textures before parenting.
        for _,child in ipairs(copy:GetChildren()) do
            if child:IsA("DataModelMesh") then
                if child:IsA("SpecialMesh") then
                    child.TextureId=""; child.VertexColor=Vector3.new(1,1,1)
                end
            else child:Destroy() end
        end
        if copy:IsA("MeshPart") then copy.TextureID="" end
        copy.Name="DaDaRoleBody"; copy.Anchored=true
        copy.CanCollide=false; copy.CanTouch=false; copy.CanQuery=false
        copy.CastShadow=false; copy.Transparency=0; copy.LocalTransparencyModifier=0
        copy.Material=Enum.Material.Neon; copy.Reflectance=0
        copy.Parent=solidWorld
        table.insert(entry.copies,{source=part,copy=copy})
    end
    local function bodySignature(parts)
        local signature={}
        for _,part in ipairs(parts) do signature[part]=part.Size end
        return signature
    end
    local function changedParts(entry,parts)
        if #parts~=entry.partCount then return true end
        for _,part in ipairs(parts) do
            if entry.signature[part]~=part.Size then return true end
        end
        return false
    end
    local function updateVolumes(player,char,hrp,col,cfg)
        local mode=(cfg.esp_cube_v1 and "1" or "0")..(cfg.esp_cube_v2 and "1" or "0")..(cfg.esp_solid_body and "1" or "0")
        if mode=="000" then dropVolumes(player); return end
        local parts=bodyParts(char)
        local entry=volumes[player]
        if entry and (entry.char~=char or entry.mode~=mode or changedParts(entry,parts)) then
            dropVolumes(player); entry=nil
        end
        if not entry then
            entry={char=char,mode=mode,objects={},boxes={},copies={},signature=bodySignature(parts),partCount=#parts}
            volumes[player]=entry
            if cfg.esp_cube_v1 then entry.outer=newBox(entry,hrp,Vector3.new(4,6,3),nil,0.72) end
            if cfg.esp_cube_v2 then
                -- Bounded voxel grid per rig part, rather than one big box per limb.
                for _,part in ipairs(parts) do
                    local size=part.Size
                    local nx=math.clamp(math.ceil(size.X/0.7),1,4)
                    local ny=math.clamp(math.ceil(size.Y/0.7),1,6)
                    local nz=math.clamp(math.ceil(size.Z/0.7),1,4)
                    local cell=Vector3.new(size.X/nx,size.Y/ny,size.Z/nz)
                    local edge=math.min(cell.X,cell.Y,cell.Z)*0.88
                    for x=1,nx do for y=1,ny do for z=1,nz do
                        local offset=CFrame.new(-size.X/2+(x-0.5)*cell.X,-size.Y/2+(y-0.5)*cell.Y,-size.Z/2+(z-0.5)*cell.Z)
                        newBox(entry,part,Vector3.new(edge,edge,edge),offset,0.18)
                    end end end
                end
            end
            if cfg.esp_solid_body then
                ensureSolidViewport()
                for _,part in ipairs(parts) do newSolidPart(entry,part) end
            end
        end
        for _,box in ipairs(entry.boxes) do box.Color3=col end
        if entry.outer then
            local minV=Vector3.new(math.huge,math.huge,math.huge)
            local maxV=Vector3.new(-math.huge,-math.huge,-math.huge)
            for _,part in ipairs(parts) do
                local half=part.Size/2
                local relative=hrp.CFrame:ToObjectSpace(part.CFrame)
                for x=-1,1,2 do for y=-1,1,2 do for z=-1,1,2 do
                    local p=relative:PointToWorldSpace(Vector3.new(half.X*x,half.Y*y,half.Z*z))
                    minV=Vector3.new(math.min(minV.X,p.X),math.min(minV.Y,p.Y),math.min(minV.Z,p.Z))
                    maxV=Vector3.new(math.max(maxV.X,p.X),math.max(maxV.Y,p.Y),math.max(maxV.Z,p.Z))
                end end end
            end
            if #parts>0 then
                entry.outer.Size=maxV-minV+Vector3.new(0.15,0.15,0.15)
                entry.outer.CFrame=CFrame.new((maxV+minV)/2)
            end
        end
        for _,pair in ipairs(entry.copies) do
            pair.copy.CFrame=pair.source.CFrame; pair.copy.Size=pair.source.Size
            pair.copy.Color=col
        end
    end

    local characterConnections = {}
    local function visit(d, fn)
        for key, object in pairs(d) do
            -- Some Drawing implementations return tables. Only Skel is a list.
            if key == "Skel" then
                for _, line in ipairs(object) do pcall(fn, line) end
            else
                pcall(fn, object)
            end
        end
    end
    local function hide_esp(d)
        visit(d, function(object) object.Visible = false end)
    end
    remove_esp = function(d)
        hide_esp(d)
        visit(d, function(object) object:Remove() end)
        table.clear(d)
    end
    local function dropHighlight(player)
        local h = highlights[player]
        highlights[player] = nil
        if h then pcall(function() h:Destroy() end) end
    end
    local function dropPlayer(player)
        local d = esp_cache[player]
        esp_cache[player] = nil
        if d then remove_esp(d) end
        dropHighlight(player)
        dropVolumes(player)
    end
    local function clearPlayers()
        for player in pairs(esp_cache) do dropPlayer(player) end
    end
    clear_highlights = function()
        for player in pairs(highlights) do dropHighlight(player) end
        clearVolumes()
    end
    local function bindPlayer(player)
        if characterConnections[player] then return end
        characterConnections[player] = player.CharacterRemoving:Connect(function()
            dropPlayer(player)
        end)
    end
    table.insert(espLifecycle.connections, Players.PlayerAdded:Connect(bindPlayer))
    table.insert(espLifecycle.connections, Players.PlayerRemoving:Connect(function(player)
        dropPlayer(player)
        local connection = characterConnections[player]
        characterConnections[player] = nil
        if connection then connection:Disconnect() end
    end))
    for _, player in ipairs(Players:GetPlayers()) do bindPlayer(player) end
    table.insert(espLifecycle.cleanups, function()
        clearPlayers()
        clear_highlights()
        for _, connection in pairs(characterConnections) do connection:Disconnect() end
        table.clear(characterConnections)
    end)
    make_esp = function(d)
        local function add(key, kind)
            local object = Drawing.new(kind)
            d[key] = object -- register before setting properties, in case an API call fails
            object.Visible = false
            return object
        end
        add("Box", "Square"); add("Name", "Text"); add("Dist", "Text")
        add("HealthBg", "Square"); add("HealthBar", "Square")
        add("Line", "Line"); add("Flag", "Text")
        d.Box.Filled=false; d.Box.Thickness=1.5
        d.Name.Center=true; d.Name.Outline=true; d.Name.Size=14
        d.Dist.Center=true; d.Dist.Outline=true; d.Dist.Size=12
        d.HealthBg.Filled=true; d.HealthBg.Color=Color3.new(0,0,0); d.HealthBg.Thickness=0
        d.HealthBar.Filled=true; d.HealthBar.Thickness=0
        d.Flag.Center=true; d.Flag.Outline=true; d.Flag.Size=11
        d.Skel={}
        for i=1,20 do
            local line=Drawing.new("Line"); d.Skel[i]=line
            line.Visible=false; line.Thickness=1; line.Transparency=1
        end
        return d
    end
    update_highlights = function(roles)
        local cam=workspace.CurrentCamera
        if not ST.esp_on or not cam then clear_highlights(); return end
        local needsSolid=false
        for _,cfg in pairs(api.profiles) do
            if cfg.enabled and cfg.esp_solid_body then needsSolid=true; break end
        end
        if needsSolid then
            ensureSolidViewport()
            solidCamera.CFrame=cam.CFrame; solidCamera.Focus=cam.Focus
            solidCamera.FieldOfViewMode=cam.FieldOfViewMode
            solidCamera.FieldOfView=cam.FieldOfView
            if cam.FieldOfViewMode==Enum.FieldOfViewMode.Diagonal then
                solidCamera.DiagonalFieldOfView=cam.DiagonalFieldOfView
            elseif cam.FieldOfViewMode==Enum.FieldOfViewMode.MaxAxis then
                solidCamera.MaxAxisFieldOfView=cam.MaxAxisFieldOfView
            end
        end
        local seen={}
        for _,player in ipairs(Players:GetPlayers()) do
            local cfg=api.profile(player,roles)
            local char=player.Character
            local hum=char and char:FindFirstChildOfClass("Humanoid")
            local hrp=char and char:FindFirstChild("HumanoidRootPart")
            if cfg.enabled and char and char:IsDescendantOf(workspace) and hum and hum.Health>0
                and hrp and (hrp.Position-cam.CFrame.Position).Magnitude<=cfg.esp_max_dist then
                seen[player]=true
                local col=api.color(api.role(player,roles),cfg)
                if cfg.esp_outline or cfg.esp_chams or cfg.esp_fill_chams then
                    local h=highlights[player]
                    if h and h.Parent~=char then dropHighlight(player); h=nil end
                    if not h then
                        h=Instance.new("Highlight"); highlights[player]=h
                        h.Name="DaDaChams"; h.Adornee=char
                        h.DepthMode=Enum.HighlightDepthMode.AlwaysOnTop; h.Parent=char
                    end
                    h.FillColor=col; h.OutlineColor=col
                    h.FillTransparency=cfg.esp_fill_chams and 0 or (cfg.esp_chams and 0.5 or 1)
                    h.OutlineTransparency=cfg.esp_outline and 0 or 1
                    h.Enabled=true
                else dropHighlight(player) end
                updateVolumes(player,char,hrp,col,cfg)
            else dropHighlight(player); dropVolumes(player) end
        end
        for player in pairs(highlights) do if not seen[player] then dropHighlight(player) end end
        for player in pairs(volumes) do if not seen[player] then dropVolumes(player) end end
        if not needsSolid then clearSolidViewport() end
    end
    esp_step = function()
        -- Fail closed: nothing from the previous frame stays visible by default.
        for player,d in pairs(esp_cache) do
            hide_esp(d)
            local char=player.Character
            local hum=char and char:FindFirstChildOfClass("Humanoid")
            if player.Parent~=Players or not char or not char:IsDescendantOf(workspace)
                or not char:FindFirstChild("HumanoidRootPart") or not hum or hum.Health<=0 then
                dropPlayer(player)
            end
        end
        if espLifecycle.stopped or not ST.esp_on then clearPlayers(); clear_highlights(); return end
        local cam=workspace.CurrentCamera
        if not cam then clear_highlights(); return end
        local roles=get_roles() or {}
        update_highlights(roles)
        for _,plr in ipairs(Players:GetPlayers()) do
            local cfg=api.profile(plr,roles)
            local char=plr.Character
            if cfg.enabled and char and char:IsDescendantOf(workspace) then
                local hrp=char:FindFirstChild("HumanoidRootPart")
                local hum=char:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health>0 then
                    local col=api.color(api.role(plr,roles),cfg)
                    local dist=(hrp.Position-cam.CFrame.Position).Magnitude
                    local d=esp_cache[plr]
                    if not d then
                        d={}; esp_cache[plr]=d
                        make_esp(d)
                    end
                    if dist>cfg.esp_max_dist then
                        hide_esp(d)
                    else
                        local cf=hrp.CFrame
                        local corners={Vector3.new(-2,2.75,-1),Vector3.new(2,2.75,-1),Vector3.new(-2,-2.75,-1),Vector3.new(2,-2.75,-1),Vector3.new(-2,2.75,1),Vector3.new(2,2.75,1),Vector3.new(-2,-2.75,1),Vector3.new(2,-2.75,1)}
                        local minX,minY=math.huge,math.huge; local maxX,maxY=-math.huge,-math.huge; local valid=true
                        for _,o in ipairs(corners) do
                            local vp,on=cam:WorldToViewportPoint((cf*CFrame.new(o)).Position)
                            if vp.Z<=0 then valid=false break end
                            if on then minX=math.min(minX,vp.X); minY=math.min(minY,vp.Y); maxX=math.max(maxX,vp.X); maxY=math.max(maxY,vp.Y) end
                        end
                        if valid and minX<maxX and minY<maxY then
                            local bw=maxX-minX; local bh=maxY-minY
                            d.Box.Size=Vector2.new(bw,bh); d.Box.Position=Vector2.new(minX,minY); d.Box.Color=col; d.Box.Visible=cfg.esp_box
                            d.Name.Text=plr.Name; d.Name.Position=Vector2.new(minX+bw/2,minY-16); d.Name.Color=col; d.Name.Visible=cfg.esp_name
                            d.Dist.Text=string.format("%dm",math.floor(dist)); d.Dist.Position=Vector2.new(minX+bw/2,maxY+2); d.Dist.Color=col; d.Dist.Visible=cfg.esp_dist
                            d.Flag.Text=api.role(plr,roles)
                            d.Flag.Position=Vector2.new(minX+bw/2,maxY+(cfg.esp_dist and 16 or 2))
                            d.Flag.Color=col; d.Flag.Visible=cfg.esp_flags==true
                            local hf=math.clamp(hum.Health/math.max(hum.MaxHealth,1),0,1)
                            d.HealthBg.Size=Vector2.new(3,bh); d.HealthBg.Position=Vector2.new(minX-6,minY); d.HealthBg.Visible=cfg.esp_health
                            d.HealthBar.Size=Vector2.new(3,bh*hf); d.HealthBar.Position=Vector2.new(minX-6,minY+bh*(1-hf)); d.HealthBar.Color=col; d.HealthBar.Visible=cfg.esp_health
                            if cfg.esp_line then d.Line.From=Vector2.new(cam.ViewportSize.X/2,cam.ViewportSize.Y); d.Line.To=Vector2.new(minX+bw/2,maxY); d.Line.Color=col; d.Line.Thickness=1; d.Line.Visible=true else d.Line.Visible=false end
                            if cfg.esp_skel then
                                local isR15=char:FindFirstChild("UpperTorso")~=nil
                                local skel=isR15 and SKEL_R15 or SKEL_R6
                                for i=1,#d.Skel do d.Skel[i].Visible=false end
                                for i,pair in ipairs(skel) do
                                    local a=char:FindFirstChild(pair[1]); local b=char:FindFirstChild(pair[2])
                                    if a and b then
                                        local ap,aon=cam:WorldToViewportPoint(a.Position); local bp,bon=cam:WorldToViewportPoint(b.Position)
                                        if aon and bon then local l=d.Skel[i]; l.From=Vector2.new(ap.X,ap.Y); l.To=Vector2.new(bp.X,bp.Y); l.Color=col; l.Visible=true end
                                    end
                                end
                            else for i=1,#d.Skel do d.Skel[i].Visible=false end end
                        else hide_esp(d) end
                    end
                end
            end
        end
    end
    -- ESP must not depend on getgun_step / silent aim succeeding in the main loop.
    local lastWarning=0
    table.insert(espLifecycle.connections, RunService.RenderStepped:Connect(function()
        if espLifecycle.stopped then return end
        local ok,err=pcall(esp_step)
        if not ok then
            clearPlayers(); clear_highlights()
            if os.clock()-lastWarning>5 then
                lastWarning=os.clock(); warn("[DaDa ESP] "..tostring(err))
            end
        end
    end))
end
initESP()

-- ============================================================
-- INIT 8: KNIFE AUTO THROW
-- ============================================================
local function initKnifeThrow()
    -- Knife-only interception. No WeaponService targeting or gun input here.
    do
        local trackers={}
        local role_data=nil
        local last_attempt,last_sent,next_roles=-math.huge,-math.huge,0
        local hook_state=nil
        local pending_input,held_input=nil,nil
        local input_token=0
        local step_busy=false
        local send_busy=false
        local vim=nil
        pcall(function() vim=game:GetService("VirtualInputManager") end)
        local stats={redirected=0,attempts=0,sent=0,errors=0,remote="none",status="off"}
        local function finite(v)
            return type(v)=="number" and v==v and math.abs(v)~=math.huge
        end
        local function number(v,fallback,lo,hi)
            v=tonumber(v)
            return math.clamp(finite(v) and v or fallback,lo,hi)
        end
        local function throwInterval()
            return number(ST.knife_interval,1,0.25,3)
        end
        local function validVector(v)
            return typeof(v)=="Vector3" and finite(v.X) and finite(v.Y) and finite(v.Z)
        end
        local function limit(v,max)
            if not validVector(v) then return Vector3.zero end
            return v.Magnitude>max and v.Unit*max or v
        end
        local function localKnife()
            local char=LP.Character
            -- A knife left in Character must not redirect an equipped gun.
            if char and char:FindFirstChild("Gun") then return nil end
            local hum=char and char:FindFirstChildOfClass("Humanoid")
            local root=char and char:FindFirstChild("HumanoidRootPart")
            local knife=char and char:FindFirstChild("Knife")
            if not hum or hum.Health<=0 or not root or not knife then return nil end
            local data=role_data and role_data.PlayerData
            local me=type(data)=="table" and data[LP.Name]
            -- Fail closed: a leftover/local cosmetic Tool does not establish a role.
            if type(me)~="table" or me.Dead or me.Role~="Murderer" then return nil end
            return knife,root
        end
        local function alive(player)
            local char=player.Character
            local hum=char and char:FindFirstChildOfClass("Humanoid")
            local part=char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"))
            if not hum or hum.Health<=0 or not part or not validVector(part.Position) then return nil end
            local data=role_data and role_data.PlayerData
            local entry=type(data)=="table" and data[player.Name]
            if entry and entry.Dead then return nil end
            return part,hum
        end
        local function nearest(root)
            local best,dist=nil,number(ST.knife_range,400,10,2000)
            for _,player in ipairs(Players:GetPlayers()) do
                if player~=LP then
                    local part=alive(player)
                    if part then
                        local d=(part.Position-root.Position).Magnitude
                        if d<=dist then best,dist=part,d end
                    end
                end
            end
            return best
        end
        local function updateTrackers(now)
            local keep={}
            for _,player in ipairs(Players:GetPlayers()) do
                if player~=LP then
                    local part=alive(player)
                    if part then
                        keep[part]=true
                        local p=part.Position
                        local engine=part.AssemblyLinearVelocity
                        local tr=trackers[part]
                        if not tr or now-tr.stamp>0.75 or (p-tr.pos).Magnitude>140 then
                            trackers[part]={pos=p,stamp=now,vel=limit(engine,170),stable=1}
                        else
                            local dt=now-tr.stamp
                            local moved=(p-tr.pos).Magnitude
                            if dt>0 and moved>0.004 then
                                local fit=limit((p-tr.pos)/dt,170)
                                local observed=validVector(engine) and limit(engine,170) or fit
                                local fresh=fit*0.7+observed*0.3
                                local a,b=Vector3.new(tr.vel.X,0,tr.vel.Z),Vector3.new(fresh.X,0,fresh.Z)
                                local turning=a.Magnitude>1 and b.Magnitude>1 and a.Unit:Dot(b.Unit)<0.5
                                tr.stable=turning and 0.35 or math.min(1,tr.stable+0.08)
                                tr.vel=tr.vel:Lerp(fresh,turning and 0.85 or 0.45)
                                tr.pos=p; tr.stamp=now
                            elseif dt>0 then
                                -- Advance the timestamp even while standing, so the
                                -- first moving sample is not averaged over idle time.
                                tr.pos=p; tr.stamp=now
                                tr.vel=validVector(engine) and limit(engine,170) or Vector3.zero
                            end
                        end
                    end
                end
            end
            for part in pairs(trackers) do if not keep[part] then trackers[part]=nil end end
        end
        local function aimPoint(part,origin)
            local base=part.Position
            if not ST.knife_predict then return base end
            local tr=trackers[part]
            local vel=tr and tr.vel or part.AssemblyLinearVelocity
            if not validVector(vel) then return base end
            -- A standing target needs no lead; leave the toggle enabled.
            local live=part.AssemblyLinearVelocity
            if validVector(live) and Vector3.new(live.X,0,live.Z).Magnitude<0.75
                and math.abs(live.Y)<1 then return base end
            local latency=ping_sec()
            latency=finite(latency) and math.clamp(latency,0,1) or 0
            local scale=number(ST.knife_lead,100,0,300)/100
            local speed=number(ST.knife_speed,96,40,400)
            local air=number(ST.knife_air,35,0,120)/100
            local extra=number(ST.knife_offset,0,-80,220)/1000
            local stable=tr and tr.stable or 1
            local hum=part.Parent and part.Parent:FindFirstChildOfClass("Humanoid")
            local airborne=hum and hum.FloorMaterial==Enum.Material.Air
            local point,travel=base,0
            for _=1,3 do
                local t=math.clamp((latency+travel)*scale+extra,0,1)
                local y=0
                if airborne and math.abs(vel.Y)<80 then
                    local tv=t*air
                    y=math.clamp(vel.Y*tv-0.5*workspace.Gravity*tv*tv,-1.8,1.8)
                end
                local offset=limit(Vector3.new(vel.X*t*stable,y,vel.Z*t*stable),60)
                point=base+offset
                travel=math.clamp((point-origin).Magnitude/speed,0,0.7)
            end
            return validVector(point) and point or base
        end
        SHF.resolveKnifeAim=function(origin)
            if espLifecycle.stopped or not ST.auto_knife_throw or getgenv().AUTOFARM_HOLD then return nil end
            local knife,root=localKnife()
            if not knife then return nil end
            local part=nearest(root)
            if not part then return nil end
            if not validVector(origin) then
                local hand=LP.Character:FindFirstChild("RightHand") or LP.Character:FindFirstChild("Right Arm")
                origin=hand and hand:IsA("BasePart") and hand.Position or root.Position
            end
            return CFrame.new(aimPoint(part,origin))
        end
        local function throwRemote()
            local knife=localKnife()
            local events=knife and knife:FindFirstChild("Events")
            local remote=events and events:FindFirstChild("KnifeThrown")
            if remote and remote:IsA("RemoteEvent") then return remote end
        end
        local function isThrow(remote)
            -- Object identity, not name matching: Shoot and all other remotes pass untouched.
            return remote~=nil and remote==throwRemote()
        end
        local function installThrowHook()
            if hook_state then return true end
            if type(hookfunction)~="function" then
                stats.status="hookfunction unavailable"
                return false
            end
            local state={active=true,busy=setmetatable({}, {__mode="k"})}
            local probe=Instance.new("RemoteEvent")
            local fire=probe.FireServer
            probe:Destroy()
            local function dispatch(previous,remote,...)
                local thread=coroutine.running()
                if state.busy[thread] or not state.active or espLifecycle.stopped
                    or not ST.auto_knife_throw or getgenv().AUTOFARM_HOLD then
                    return previous(remote,...)
                end
                local args=table.pack(...)
                local ok,target=pcall(function()
                    if not isThrow(remote) or typeof(args[1])~="CFrame"
                        or typeof(args[2])~="CFrame" then return nil end
                    return SHF.resolveKnifeAim(args[1].Position)
                end)
                if not ok then
                    stats.errors=stats.errors+1; stats.status=tostring(target)
                    return previous(remote,...)
                end
                if not target then return previous(remote,...) end
                -- Both hook entry points and the automatic sender share one cooldown.
                -- Reserve before sending: a yielding/erroring send cannot admit duplicates.
                local now=os.clock()
                if send_busy or now-last_sent<throwInterval() then
                    if pending_input then pending_input.seen=true end
                    stats.status="throw cooldown"
                    return
                end
                last_sent=now
                send_busy=true
                -- Consume the original request. Send exactly one replacement through
                -- the unhooked function; preserve A, trailing arguments and nils.
                args[2]=target
                if pending_input then pending_input.seen=true end
                state.busy[thread]=true
                local result=table.pack(pcall(state.fireOriginal,remote,table.unpack(args,1,args.n)))
                state.busy[thread]=nil
                send_busy=false
                if not result[1] then
                    stats.errors=stats.errors+1; stats.status=tostring(result[2])
                    -- Never retry here: a retry could duplicate an already sent event.
                    error(result[2],0)
                end
                last_sent=os.clock()
                stats.redirected=stats.redirected+1
                stats.sent=stats.sent+1; stats.remote="KnifeThrown"; stats.status="redirected"
                return table.unpack(result,2,result.n)
            end
            local function fireWrapper(remote,...)
                return dispatch(state.fireOriginal,remote,...)
            end
            local wrap=type(newcclosure)=="function" and newcclosure(fireWrapper) or fireWrapper
            local ok,previous=pcall(hookfunction,fire,wrap)
            if not ok or type(previous)~="function" then
                state.active=false; stats.status="FireServer hook failed: "..tostring(previous)
                return false
            end
            state.fireOriginal=previous
            state.fireWrapper=wrap
            hook_state=state
            -- Some executors route colon calls through __namecall without visiting
            -- the hooked function. Both entry points use the same guarded sender.
            if type(hookmetamethod)=="function" and type(getnamecallmethod)=="function" then
                local function namecallWrapper(remote,...)
                    if getnamecallmethod()~="FireServer" then return state.nameOriginal(remote,...) end
                    return dispatch(state.nameOriginal,remote,...)
                end
                local nw=type(newcclosure)=="function" and newcclosure(namecallWrapper) or namecallWrapper
                local nok,np=pcall(hookmetamethod,game,"__namecall",nw)
                if nok and type(np)=="function" then state.nameOriginal=np; state.nameWrapper=nw end
            end
            stats.status="hook ready"
            return true
        end
        local function sendInput(down,mode,backend)
            if mode=="E" then
                if backend=="executor" then
                    if down then keypress(0x45) else keyrelease(0x45) end
                else vim:SendKeyEvent(down,Enum.KeyCode.E,false,game) end
            else
                if backend=="executor" then
                    if down then mouse2press() else mouse2release() end
                else
                    local mouse=UIS:GetMouseLocation()
                    vim:SendMouseButtonEvent(mouse.X,mouse.Y,1,down,game,0)
                end
            end
        end
        local function releaseInput()
            local held=held_input
            held_input=nil
            if held then pcall(sendInput,false,held.mode,held.backend) end
        end
        local function cancelInput()
            input_token=input_token+1
            releaseInput(); pending_input=nil
        end
        local function pressInput()
            local mode=ST.knife_input=="Right Mouse" and "Right Mouse" or "E"
            local native=mode=="E" and type(keypress)=="function" and type(keyrelease)=="function"
                or mode=="Right Mouse" and type(mouse2press)=="function" and type(mouse2release)=="function"
            local backend=native and "executor" or (vim and "vim" or nil)
            if not backend then stats.status="input unavailable; select Remote mode"; return false end
            local ok,err=pcall(sendInput,true,mode,backend)
            if not ok then
                pcall(sendInput,false,mode,backend)
                stats.errors=stats.errors+1; stats.status=tostring(err); return false
            end
            held_input={mode=mode,backend=backend}
            input_token=input_token+1
            local token=input_token
            task.delay(number(ST.knife_hold,0.15,0.05,0.5),function()
                if token==input_token then releaseInput() end
            end)
            return true
        end
        SHF.setKnifeThrowing=function(enabled)
            if not enabled then
                ST.auto_knife_throw=false; stats.status="off"
                cancelInput(); table.clear(trackers)
                return true
            end
            if espLifecycle.stopped or not installThrowHook() then
                ST.auto_knife_throw=false
                if notify then notify("Knife",stats.status) end
                return false
            end
            ST.auto_knife_throw=true; stats.status="waiting for equipped knife"
            -- Preserve the cooldown when toggled off/on.
            last_attempt=math.max(last_attempt,last_sent)
            return true
        end
        SHF.knifeDiagnostics=function()
            if notify then
                notify("Knife",string.format("function hook: %s | namecall: %s | replaced: %d | sent: %d | errors: %d | %s",
                    hook_state and "on" or "off",hook_state and hook_state.nameOriginal and "on" or "off",
                    stats.redirected,stats.sent,stats.errors,stats.status))
            end
        end
        local function step()
            if espLifecycle.stopped or not ST.auto_knife_throw then
                if held_input or pending_input then cancelInput() end
                return
            end
            local now=os.clock()
            if now>=next_roles then
                next_roles=now+0.5
                local modules=ReplicatedStorage:FindFirstChild("Modules")
                local module=modules and modules:FindFirstChild("CurrentRoundClient")
                if not role_data and module then
                    local ok,data=pcall(require,module)
                    if ok and type(data)=="table" then role_data=data end
                end
            end
            if not installThrowHook() then return end
            local knife,root=localKnife()
            if getgenv().AUTOFARM_HOLD or not knife then
                cancelInput(); stats.status="waiting for equipped knife"; return
            end
            updateTrackers(now)
            if pending_input and now>=pending_input.deadline then
                if not pending_input.seen then stats.status="no game throw event; try Remote mode" end
                pending_input=nil
            end
            if ST.knife_auto_mode~="Game Input" and (held_input or pending_input) then cancelInput() end
            if pending_input or held_input or UIS:GetFocusedTextBox() then return end
            if send_busy or now-math.max(last_attempt,last_sent)<throwInterval() then return end
            local remote=throwRemote()
            if not remote then stats.status="Knife.Events.KnifeThrown not found"; return end
            local char=LP.Character
            local hand=char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm")
            local origin=hand and hand:IsA("BasePart") and hand.CFrame or root.CFrame
            local target=SHF.resolveKnifeAim(origin.Position)
            if not target then stats.status="no live target in range"; return end
            last_attempt=now; stats.attempts=stats.attempts+1
            if ST.knife_auto_mode=="Game Input" then
                pending_input={seen=false,deadline=now+math.max(0.8,number(ST.knife_hold,0.15,0.05,0.5)+0.5)}
                if not pressInput() then pending_input=nil end
            else
                -- Direct automatic throw, independent of input and gun hooks.
                -- Native game input remains available above for local tool animations.
                local thread=coroutine.running()
                -- Recheck after aim resolution before bypassing the hooks.
                if not localKnife() or send_busy or os.clock()-last_sent<throwInterval() then return end
                send_busy=true
                last_sent=os.clock()
                hook_state.busy[thread]=true
                local ok,err=pcall(hook_state.fireOriginal,remote,origin,target)
                hook_state.busy[thread]=nil
                send_busy=false
                if ok then
                    stats.sent=stats.sent+1
                    stats.remote="KnifeThrown"; stats.status="auto remote sent (server acceptance unverified)"
                else stats.errors=stats.errors+1; stats.status=tostring(err) end
            end
        end
        -- Dedicated heartbeat: errors in the old combat loop cannot stop throwing.
        table.insert(espLifecycle.connections,RunService.Heartbeat:Connect(function()
            -- require/input calls may yield; do not stack Heartbeat workers.
            if step_busy then return end
            step_busy=true
            local ok,err=pcall(step)
            step_busy=false
            if not ok then
                cancelInput(); stats.errors=stats.errors+1; stats.status=tostring(err)
            end
        end))
        knife_silent_step=nil
        table.insert(espLifecycle.connections,LP.CharacterRemoving:Connect(function()
            cancelInput(); table.clear(trackers)
        end))
        table.insert(espLifecycle.connections,LP.CharacterAdded:Connect(function()
            cancelInput(); table.clear(trackers); last_attempt=-math.huge
        end))
        table.insert(espLifecycle.cleanups,function()
            SHF.setKnifeThrowing(false)
            -- Leave inert pass-throughs rather than overwrite a later hook chain.
            if hook_state then hook_state.active=false end
        end)

    end
end

-- ============================================================
-- INDEPENDENT AIM LOCK + IMAGE ROLE ARROWS
-- ============================================================
local function initAimLockAndArrows()
    local roundData=nil
    local roster={}
    local nextRefresh=0
    local arrows={}
    local arrowsGui=nil
    local arrowImages={
        ["normal"]="rbxassetid://118079146241188",
        ["big bon"]="rbxassetid://118171553302006",
        ["big hun"]="rbxassetid://79417965357382",
        ["big hon"]="rbxassetid://125861517297169",
        ["White triangle"]="rbxassetid://98993244838876",
        ["Black triangle"]="rbxassetid://72740468391101",
    }
    local cameraBinding="DaDa_AimLock_"..tostring(LP.UserId)
    local white=Color3.fromRGB(255,255,255)
    local red=Color3.fromRGB(255,60,60)
    local blue=Color3.fromRGB(0,153,255)
    local function refreshRoster()
        if not roundData then
            local modules=ReplicatedStorage:FindFirstChild("Modules")
            local module=modules and modules:FindFirstChild("CurrentRoundClient")
            if module then
                local ok,data=pcall(require,module)
                if ok and type(data)=="table" then roundData=data end
            end
        end
        local fresh={}
        local data=roundData and roundData.PlayerData
        for _,player in ipairs(Players:GetPlayers()) do
            if player~=LP then
                local info=type(data)=="table" and data[player.Name]
                local role=type(info)=="table" and info.Role or "Innocent"
                local char=player.Character
                local bag=player:FindFirstChildOfClass("Backpack")
                if holds(char,"Knife") or holds(bag,"Knife") then role="Murderer"
                elseif role~="Murderer" and (holds(char,"Gun") or holds(bag,"Gun")) then role="Sheriff" end
                fresh[#fresh+1]={player=player,role=role,dead=type(info)=="table" and info.Dead==true}
            end
        end
        roster=fresh
    end
    local function alive(entry)
        if entry.dead or entry.player.Parent~=Players then return nil end
        local char=entry.player.Character
        local hum=char and char:FindFirstChildOfClass("Humanoid")
        local part=char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"))
        if hum and hum.Health>0 and part and part:IsA("BasePart") then return part end
    end
    local function clearPlayer(player)
        local image=arrows[player]
        if image then image:Destroy(); arrows[player]=nil end
    end
    local function clearAll()
        local keys={}
        for player in pairs(arrows) do keys[#keys+1]=player end
        for _,player in ipairs(keys) do clearPlayer(player) end
        if arrowsGui then arrowsGui.Enabled=false end
    end
    local function ensureGui()
        if arrowsGui and arrowsGui.Parent then return arrowsGui end
        clearAll()
        if arrowsGui then arrowsGui:Destroy(); arrowsGui=nil end
        local screen=Instance.new("ScreenGui")
        screen.Name="DaDa_RoleArrows"
        screen.IgnoreGuiInset=true
        screen.ResetOnSpawn=false
        screen.DisplayOrder=80
        screen.ZIndexBehavior=Enum.ZIndexBehavior.Sibling
        local parents={}
        if type(gethui)=="function" then
            local ok,parent=pcall(gethui)
            if ok and parent then parents[#parents+1]=parent end
        end
        if CoreGui then parents[#parents+1]=CoreGui end
        local playerGui=LP:FindFirstChildOfClass("PlayerGui")
        if playerGui then parents[#parents+1]=playerGui end
        local attached=false
        for _,parent in ipairs(parents) do
            local ok=pcall(function() screen.Parent=parent end)
            if ok and screen.Parent then attached=true; break end
        end
        if not attached then screen:Destroy(); error("Arrows GUI parent unavailable") end
        arrowsGui=screen
        return screen
    end
    local function makeArrow(player,parent)
        local arrowImage=arrowImages[ST.arrows_style] or arrowImages["normal"]
        local image=arrows[player]
        if image and image.Parent==parent then
            if image.Image~=arrowImage then image.Image=arrowImage end
            return image
        end
        if image then image:Destroy() end
        image=Instance.new("ImageLabel")
        image.Name="Arrow_"..tostring(player.UserId)
        image.BackgroundTransparency=1
        image.AnchorPoint=Vector2.new(0.5,0.5)
        image.Image=arrowImage
        image.ImageTransparency=0
        image.ScaleType=Enum.ScaleType.Fit
        image.Active=false
        image.ZIndex=1
        image.Visible=false
        image.Parent=parent
        arrows[player]=image
        return image
    end
    local function arrowsStep(camera,dt)
        if not ST.arrows then
            if next(arrows) then clearAll() end
            if arrowsGui then arrowsGui.Enabled=false end
            return
        end
        local viewport=camera.ViewportSize
        if viewport.X<2 or viewport.Y<2 then
            if arrowsGui then arrowsGui.Enabled=false end
            return
        end
        local parent=ensureGui()
        parent.Enabled=true
        local center=viewport*0.5
        local size=math.clamp(tonumber(ST.arrows_size) or 26,12,64)
        -- Reserve half the image diagonal so rotation stays inside the viewport.
        local maxRadius=math.max(0,math.min(viewport.X,viewport.Y)*0.5-size*0.7072-6)
        local radius=math.clamp(tonumber(ST.arrows_radius) or 160,0,maxRadius)
        local hideOnScreen=ST.arrows_mode=="Hide On Screen"
        local fade=1-math.exp(-18*math.clamp(tonumber(dt) or 1/60,0,0.1))
        local used={}
        for _,entry in ipairs(roster) do
            local part=alive(entry)
            if part then
                local relative=camera.CFrame:PointToObjectSpace(part.Position)
                -- Camera forward is -Z; screen up is -Y. Do not negate Z:
                -- doing so swaps front/back and mirrors the arrow vertically.
                local bearing=Vector2.new(relative.X,relative.Z)
                local dir=bearing.Magnitude>0.001 and bearing.Unit or Vector2.new(0,-1)
                local anchor=center+dir*radius
                -- Resolve the same live role profile as ESP; do not cache its color.
                local api=ST._espRoles
                local color
                if api then
                    local role=api.role(entry.player,{[entry.player.Name]=entry.role})
                    color=api.color(role,api.profiles[role])
                else
                    color=entry.role=="Murderer" and red
                        or ((entry.role=="Sheriff" or entry.role=="Hero") and blue or white)
                end
                local image=makeArrow(entry.player,parent)
                image.Size=UDim2.fromOffset(size,size)
                image.Position=UDim2.fromOffset(anchor.X,anchor.Y)
                -- The source arrow is treated as pointing up at zero rotation.
                image.Rotation=math.deg(math.atan2(dir.Y,dir.X))+90
                image.ImageColor3=color
                if hideOnScreen then
                    local screenPoint,onScreen=camera:WorldToViewportPoint(part.Position)
                    local targetTransparency=(onScreen and screenPoint.Z>0) and 1 or 0
                    local transparency=image.ImageTransparency
                    transparency=transparency+(targetTransparency-transparency)*fade
                    if math.abs(transparency-targetTransparency)<0.005 then
                        transparency=targetTransparency
                    end
                    image.ImageTransparency=math.clamp(transparency,0,1)
                else
                    -- Switching back to Always must also restore faded arrows.
                    image.ImageTransparency=0
                end
                image.Visible=image.ImageTransparency<1
                used[entry.player]=true
            end
        end
        local stale={}
        for player in pairs(arrows) do if not used[player] then stale[#stale+1]=player end end
        for _,player in ipairs(stale) do clearPlayer(player) end
    end
    local function lockStep(camera)
        if not ST.aim_lock or getgenv().AUTOFARM_HOLD or UIS:GetFocusedTextBox() then return end
        local char=LP.Character
        local hum=char and char:FindFirstChildOfClass("Humanoid")
        local root=char and char:FindFirstChild("HumanoidRootPart")
        if not root or not hum or hum.Health<=0 then return end
        local selected=nil
        local distance=math.huge
        for _,entry in ipairs(roster) do
            local role=entry.role
            local filter=ST.aim_lock_target
            if filter=="Nearest" or (filter=="Sheriff" and (role=="Sheriff" or role=="Hero"))
                or (filter=="Murder" and role=="Murderer") then
                local part=alive(entry)
                if part then
                    local d=(part.Position-root.Position).Magnitude
                    if d<distance then distance=d; selected=part end
                end
            end
        end
        if not selected then return end
        local point=selected.Position
        if ST.aim_lock_predict then
            local vel=selected.AssemblyLinearVelocity
            local horizontal=Vector3.new(vel.X,0,vel.Z)
            if horizontal.Magnitude>=0.75 then
                if horizontal.Magnitude>170 then horizontal=horizontal.Unit*170 end
                point=point+horizontal*math.clamp(tonumber(ST.aim_lock_lead) or 0.12,0,0.5)
            end
        end
        local origin=camera.CFrame.Position
        if (point-origin).Magnitude>0.01 then camera.CFrame=CFrame.lookAt(origin,point) end
    end
    table.insert(espLifecycle.connections,RunService.Heartbeat:Connect(function()
        if espLifecycle.stopped or not (ST.aim_lock or ST.arrows) then return end
        local now=os.clock()
        if now>=nextRefresh then
            nextRefresh=now+0.2
            local ok,err=pcall(refreshRoster)
            if not ok then SHF.aim_arrows_error=tostring(err) end
        end
    end))
    -- Runs after the default camera controller so it cannot overwrite the lock.
    RunService:BindToRenderStep(cameraBinding,Enum.RenderPriority.Camera.Value+1,function(dt)
        if espLifecycle.stopped then return end
        local camera=workspace.CurrentCamera
        if not camera then return end
        local ok,err=pcall(lockStep,camera)
        if not ok then SHF.aim_lock_error=tostring(err) end
        ok,err=pcall(arrowsStep,camera,dt)
        if not ok then
            SHF.arrows_error=tostring(err); ST.arrows=false; clearAll()
            if notify then notify("Arrows",tostring(err)) end
        end
    end)
    table.insert(espLifecycle.connections,Players.PlayerRemoving:Connect(clearPlayer))
    table.insert(espLifecycle.cleanups,function()
        ST.aim_lock=false; ST.arrows=false
        RunService:UnbindFromRenderStep(cameraBinding)
        clearAll(); table.clear(roster)
        if arrowsGui then arrowsGui:Destroy(); arrowsGui=nil end
    end)
end
initAimLockAndArrows()

-- INIT 9: FARM + FLING + ANTIFLING + KILL AURA

local function initFarmFling()
    local was_down=false; local down_ref_y=nil; local last_touch=0
    local farm_target=nil; local collected_ids={}; local collected_count=0
    local coins_done=false; local saw_coins=false
    getgenv().AUTOFARM_HOLD=false
    farm_hrp=function() local c=LP.Character; return c and c:FindFirstChild("HumanoidRootPart") end
    local function get_round_data() local m=get_round(); return m and m.PlayerData end
    can_farm=function()
        local char=LP.Character; local hum=char and char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health<=0 then return false end
        local d=get_round_data()
        if type(d)=="table" then local me=d[LP.Name]; if not me or not me.Role or me.Dead then return false end end
        return true
    end
    coin_bags_full=function()
        local pg=LP:FindFirstChild("PlayerGui"); local main=pg and pg:FindFirstChild("MainGUI"); local gg=main and main:FindFirstChild("Game")
        local bags=gg and gg:FindFirstChild("CoinBags"); local container=bags and bags:FindFirstChild("Container")
        if not container then return false end
        local any=false
        for _,v in ipairs(container:GetChildren()) do
            if v:IsA("Frame") and v.Visible then
                any=true
                local full=v:FindFirstChild("Full")
                if not (full and full.Visible) then return false end
            end
        end
        return any
    end
    local function coin_available(v) return v and v.Parent and v:IsA("BasePart") and not v:GetAttribute("Collected") and not v:GetAttribute("Delete") end
    local function nearest_coin(pos,list)
        local best,bd=nil,math.huge
        for _,v in ipairs(list) do
            local d=(v.Position-pos).Magnitude
            if d<bd then bd=d; best=v end
        end
        return best
    end
    local function set_farm_noclip(on)
        local char=LP.Character; local hum=char and char:FindFirstChildOfClass("Humanoid")
        if on then
            if not char then return end
            if hum then pcall(function() hum.PlatformStand=true end) end
            for _,p in ipairs(char:GetDescendants()) do
                if p:IsA("BasePart") and p.CanCollide then
                    if nc_cache[p]==nil then nc_cache[p]=p.CanCollide end
                    p.CanCollide=false
                end
            end
        else
            if hum then pcall(function() hum.PlatformStand=false end) end
            for p,v in pairs(nc_cache) do if p and p.Parent then pcall(function() p.CanCollide=v end) end end
            nc_cache={}
        end
    end
    local up_params=RaycastParams.new()
    up_params.FilterType=Enum.RaycastFilterType.Exclude
    up_params.IgnoreWater=true
    local function return_to_surface()
        local hrp=farm_hrp(); if not hrp then return end
        local origin=hrp.Position
        up_params.FilterDescendantsInstances={LP.Character}
        local res=workspace:Raycast(origin,Vector3.new(0,400,0),up_params)
        local y=res and (res.Position.Y+5) or (down_ref_y and down_ref_y+5 or nil)
        if not y then return end
        pcall(function()
            hrp.CFrame=CFrame.new(origin.X,y,origin.Z)
            hrp.AssemblyLinearVelocity=Vector3.zero
            hrp.AssemblyAngularVelocity=Vector3.zero
        end)
    end
    farm_release=function() if was_down then was_down=false; return_to_surface() end; set_farm_noclip(false) end
    local murder_hrp_cache,murder_hrp_t=nil,0
    local function murderer_hrp()
        local now=os.clock()
        if now-murder_hrp_t<0.25 then return murder_hrp_cache end
        murder_hrp_t=now; murder_hrp_cache=nil
        local d=get_round_data()
        if type(d)=="table" then
            local me=d[LP.Name]
            if me and me.Role=="Murderer" then return nil end
            for name,info in pairs(d) do
                if type(info)=="table" and info.Role=="Murderer" and not info.Dead and name~=LP.Name then
                    local pl=Players:FindFirstChild(name)
                    local char=pl and pl.Character
                    local h=char and char:FindFirstChild("HumanoidRootPart")
                    local hum=char and char:FindFirstChildOfClass("Humanoid")
                    if h and (not hum or hum.Health>0) then murder_hrp_cache=h end
                    break
                end
            end
        end
        return murder_hrp_cache
    end
    local function flat_dist(a,b) return Vector3.new(a.X-b.X,0,a.Z-b.Z).Magnitude end
    local function pick_coin(pos,list,mpos)
        if not (farm_cfg.avoid_murder and mpos) then return nearest_coin(pos,list) end
        local safe,sd=nil,math.huge
        local far,fd=nil,-1
        for _,v in ipairs(list) do
            local md=flat_dist(v.Position,mpos)
            if md>fd then fd=md; far=v end
            if md>=farm_cfg.avoid_dist then
                local d=(v.Position-pos).Magnitude
                if d<sd then sd=d; safe=v end
            end
        end
        if safe then return safe end
        if far and fd>=farm_cfg.avoid_dist*0.6 then return far end
        return nil
    end
    local function coin_ok_now(v,mpos)
        if not coin_available(v) then return false end
        if farm_cfg.avoid_murder and mpos and flat_dist(v.Position,mpos)<farm_cfg.avoid_dist*0.6 then return false end
        return true
    end
    local function avoid_steer(cur,dest,mpos)
        if not (farm_cfg.avoid_murder and mpos) then return dest end
        local dm=flat_dist(cur,mpos)
        if dm>=farm_cfg.avoid_dist then return dest end
        local away=Vector3.new(cur.X-mpos.X,0,cur.Z-mpos.Z)
        if away.Magnitude<0.1 then away=Vector3.new(1,0,0) end
        away=away.Unit
        local want=Vector3.new(dest.X-cur.X,0,dest.Z-cur.Z)
        local mag=want.Magnitude
        if mag<0.1 then return dest end
        local weight=1+(1-dm/farm_cfg.avoid_dist)*2
        local blend=(want.Unit+away*weight)
        if blend.Magnitude<0.1 then blend=away else blend=blend.Unit end
        local np=cur+blend*mag
        return Vector3.new(np.X,dest.Y,np.Z)
    end
    local function touch_targets(coin)
        local list,seen={},{}
        local function add(p)
            if p and not seen[p] and p:IsA("BasePart") and p:FindFirstChildOfClass("TouchTransmitter") then
                seen[p]=true; list[#list+1]=p
            end
        end
        add(coin)
        for _,v in ipairs(coin:GetChildren()) do add(v) end
        local par=coin.Parent
        if par then
            if par:IsA("BasePart") then add(par) end
            for _,v in ipairs(par:GetChildren()) do add(v) end
        end
        if #list==0 then list[1]=coin end
        return list
    end
